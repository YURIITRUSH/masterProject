wait 2
status

# 1. Базова лінія (Open-Loop)
f 100
wait 2

# 2. Запуск PBC
w 10
wait 5

# 3. Прискорення
w 15
wait 5

# 4. Активне гальмування (найцікавіше)
w 5
wait 8  # Даємо більше часу, щоб побачити, як PBC гасить інерцію і долає тертя

# 5. Знову прискорення
w 15
wait 5

# 6. Реверс через нуль
w -10
wait 8

# 7. Стоп
s
wait 1
stop_all
